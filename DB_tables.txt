create table print_jobs (
  si_no serial primary key,
  hall_id text,
  logged_user boolean not null default false,
  payment_method text not null check (payment_method in ('direct','wallet')),
  txn_id text,
  amount_paid numeric(6,2) check (amount_paid > 0),
  sender_number text,
  amount_calculated numeric(6,2) not null check (amount_calculated > 0),
  files jsonb not null,
  total_files int not null,
  total_page_print int not null,
  job_ids text[],
  status boolean not null default false,
  comments text,
  created_at timestamptz default now()
);






Table-03: payments
1. Si_No (PK)= (auto gen, koto number print eita: 1,2,3,4,.....)
2. txn_id	(text)=	bKash/Nagad txn id
3. via (text)= bkash / nagad / rocket / bangla_qr
4. sender_number (text)=	কে পাঠিয়েছে
5. amount	(numeric)=	কত টাকা
6. status	(text)=	not_used / used
7. use_for: (direct Print / Top-up)
8. print_job_si_no (FK)=	কোন print job এ use হয়েছে
9. created_at	(timestamptz)=	কখন payment আসলো

create table payments (
  si_no serial primary key,
  via text not null,
  txn_id text unique not null,
  sender_number text not null,
  amount numeric(6,2) not null
    check (amount > 0),
  status text not null default 'not_used'
    check (status in ('not_used', 'used')),
  use_for text
    check (use_for in ('direct_print', 'top_up')),
  print_job_si_no int references print_jobs(si_no),
  created_at timestamptz default now()
);
