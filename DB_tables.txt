-- TABLE 2: print_jobs
-- প্রতিটা print request এর main record
-- ════════════════════════════════════════════════════════════

create table print_jobs (
  si_no            serial primary key,
  hall_id          text,
  logged_user      boolean not null default false,
  payment_method   text not null check (payment_method in ('direct', 'wallet')),
  txn_id           text,
  amount_paid      numeric(6,2) check (amount_paid > 0),
  sender_number    text,
  amount_calculated numeric(6,2) not null check (amount_calculated > 0),
  files            jsonb not null,        -- [{name, pages, copies, color, subtotal}]
  total_files      int not null,
  total_page_print int not null,
  job_ids          text[],               -- Epson cloud job IDs
  status           boolean not null default false,  -- true = Epson এ পাঠানো সফল
  comments         text,                 -- overpaid / short / null
  created_at       timestamptz default now()
);

alter table print_jobs enable row level security;
create policy "service role only" on print_jobs using (false);




-- ════════════════════════════════════════════════════════════
-- TABLE 3: mfs_transactions
-- bKash / Nagad SMS parser থেকে আসা real-time transaction data
-- payments table এর বদলে এটাই use হচ্ছে
-- ════════════════════════════════════════════════════════════

create table if not exists public.mfs_transactions (
  id                     uuid primary key default gen_random_uuid(),
  provider               varchar(20) not null,       -- 'bKash' | 'Nagad'
  transaction_type       varchar(50) not null,        -- 'receiveMoney' | 'Receive Money' etc.
  amount                 numeric(12,2),
  currency               varchar(10) default 'BDT',
  trx_id                 varchar(50) unique,          -- e.g. '9K2L1M4N'
  counterparty_role      varchar(50),                 -- 'sender' | 'recipient' etc.
  counterparty_identifier varchar(50),                -- '017XXXXXXXX'
  counterparty_name      varchar(100),
  balance                numeric(12,2),               -- transaction পরে remaining balance
  fee                    numeric(12,2),
  reference              text,
  sms_timestamp          varchar(50),                 -- SMS এর raw timestamp
  raw_text               text not null,               -- original SMS string
  confidence_score       numeric(4,3) default 1.0,    -- parser accuracy
  device_id              varchar(100),                -- Android device ID
  status                 text not null default 'not_used'
                           check (status in ('not_used', 'used')),
  use_for                text
                           check (use_for in ('direct_print', 'top_up')),
  received_at            timestamptz default now(),
  created_at             timestamptz default now()
);

-- Indexes
create index if not exists idx_mfs_trx_id    on public.mfs_transactions(trx_id);
create index if not exists idx_mfs_provider  on public.mfs_transactions(provider);
create index if not exists idx_mfs_type      on public.mfs_transactions(transaction_type);
create index if not exists idx_mfs_created_at on public.mfs_transactions(created_at desc);

-- RLS
alter table public.mfs_transactions enable row level security;

create policy "Allow public insert" on public.mfs_transactions
  for insert with check (true);

create policy "Allow public read" on public.mfs_transactions
  for select using (true);




-- ════════════════════════════════════════════════════════════
-- TABLE 4: user_wallets
-- Supabase auth.users এর সাথে linked wallet
-- ════════════════════════════════════════════════════════════

create table user_wallets (
  user_id    uuid primary key references auth.users(id) on delete cascade,
  roll       bigint unique,
  email      text unique,
  balance    numeric(6,2) default 0 check (balance >= 0),
  updated_at timestamptz default now()
);

alter table user_wallets enable row level security;
create policy "service role only" on user_wallets using (false);

-- FUNCTIONS
-- ════════════════════════════════════════════════════════════

-- Wallet থেকে balance কাটো — atomic, race condition safe
create or replace function deduct_wallet_balance(
  p_user_id uuid,
  p_amount  numeric
)
returns numeric
language plpgsql
as $$
declare
  v_current numeric;
  v_new     numeric;
begin
  select balance into v_current
  from user_wallets
  where user_id = p_user_id
  for update;

  if v_current is null then
    raise exception 'Wallet not found';
  end if;

  if v_current < p_amount then
    raise exception 'Insufficient balance. Available: %', v_current;
  end if;

  v_new := round(v_current - p_amount, 2);

  update user_wallets
  set balance = v_new
  where user_id = p_user_id;

  return v_new;
end;
$$;


-- Wallet এ balance যোগ করো — atomic, race condition safe
create or replace function add_wallet_balance(
  p_user_id uuid,
  p_amount  numeric
)
returns numeric
language plpgsql
as $$
declare
  v_new numeric;
begin
  update user_wallets
  set balance = round(balance + p_amount, 2)
  where user_id = p_user_id
  returning balance into v_new;

  if v_new is null then
    raise exception 'Wallet not found';
  end if;

  return v_new;
end;
$$;
